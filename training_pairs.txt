### EXAMPLE 1
INPUT:
TASK: Create a Manim animation.

OBJECTS:
- A blue circle at position (0, 0) with radius 1.

ANIMATION STEPS:
1. Show the blue circle at the center of the screen.
2. Move the circle smoothly to position (3, 0) over 2 seconds.

OUTPUT:
from manim import *

class AutoScene(Scene):
    def construct(self):
        circle = Circle(radius=1.0, color=BLUE).move_to([0, 0])
        self.add(circle)
        self.play(circle.animate.move_to([3, 0]), run_time=2.0)

### EXAMPLE 4
INPUT:
TASK: Create a Manim animation.

OBJECTS:
- A blue circle at position (-2, 0) with radius 0.7.
- A green circle at position (2, 0) with radius 0.7.

ANIMATION STEPS:
1. Show both circles in their starting positions.
2. Over 2 seconds, smoothly move the blue circle to (2, 0) and the green circle to (-2, 0), as if they swap places.
3. Hold the final positions for 1 second.

OUTPUT:
from manim import *

class AutoScene(Scene):
    def construct(self):
        left_circle = Circle(radius=0.7, color=BLUE).move_to([-2, 0, 0])
        right_circle = Circle(radius=0.7, color=GREEN).move_to([2, 0, 0])

        self.play(FadeIn(left_circle), FadeIn(right_circle))
        self.play(
            left_circle.animate.move_to([2, 0, 0]),
            right_circle.animate.move_to([-2, 0, 0]),
            run_time=2.0
        )
        self.wait(1)

### EXAMPLE 6
INPUT:
TASK: Create a Manim animation.

OBJECTS:
- A set of 2D axes.
- A red dot that moves along the x-axis from x = -2 to x = 2.

ANIMATION STEPS:
1. Create and show the axes.
2. Show the red dot at position (-2, 0).
3. Over 3 seconds, move the dot smoothly along the x-axis to position (2, 0).
4. Hold for 1 second.

OUTPUT:
from manim import *

class AutoScene(Scene):
    def construct(self):
        axes = Axes(x_range=[-3, 3, 1], y_range=[-2, 2, 1])
        dot = Dot(color=RED).move_to(axes.coords_to_point(-2, 0))

        self.play(Create(axes))
        self.play(FadeIn(dot))
        self.play(dot.animate.move_to(axes.coords_to_point(2, 0)), run_time=3.0)
        self.wait(1)
### EXAMPLE 7
INPUT:
TASK: Create a Manim animation.

OBJECTS:
- A white text saying "Step 1" at the center of the screen.
- A green text saying "Step 2" at the center of the screen.

ANIMATION STEPS:
1. Start with a blank screen.
2. Write "Step 1" at the center over 1 second.
3. Hold "Step 1" for 1 second.
4. Transform "Step 1" into "Step 2" over 1.5 seconds.
5. Hold "Step 2" for 1 second.

OUTPUT:
from manim import *

class AutoScene(Scene):
    def construct(self):
        step1 = Text("Step 1", color=WHITE)
        step2 = Text("Step 2", color=GREEN)

        self.play(Write(step1), run_time=1.0)
        self.wait(1)
        self.play(Transform(step1, step2), run_time=1.5)
        self.wait(1)
